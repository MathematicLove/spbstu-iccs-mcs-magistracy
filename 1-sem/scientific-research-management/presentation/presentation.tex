% - Введение -
\begin{frame}{Введение}
  \small
  \begin{itemize}
    \item CNN: ограниченное рецептивное поле, нет глобального контекста, сегментация сводится к классификации пикселей
    \item ViT: глобальный контекст за счёт самовнимания, но признаки одномасштабные
    \item Решение ?: SegFormer - многоуровневые признаки, Efficient Self-Attention снижает сложность до $O(N^2/R)$, но она остаётся квадратичной
    \item Научная проблема - точность глобального внимания против его квадратичной сложности, мешающей работе в реальном времени
    \item Объект - сегментаторы на основе ViT; предмет - методы линеаризации механизма внимания
    \item Цель - найти метод линеаризации, сохраняющий точность сегментации и обеспечивающий работу в реальном времени
    \item Задачи - анализ методов линеаризации, сравнение по точности и быстродействию, адаптация к многоуровневой архитектуре, экспериментальная апробация
    \item Применение - КТ и другие медицинские снимки, дорожные сцены, контроль дефектов, динамическая зона интереса трамвайных путей~\cite{myarticle}
  \end{itemize}
\end{frame}

% - Проблемы современных ViT -
\begin{frame}{Проблемы современных ViT}
  \small
  \begin{itemize}
    \item Самовнимание: $\mathrm{softmax}\left(QK^{\top} / \sqrt{d}\right) V$, время $O(N^2 d)$, память $O(N^2)$
    \item При шаге 4 и кадре $1024 \times 1024$: $N = 65536$, матрица весов $4.3 \cdot 10^{9}$ элементов, около 17 ГБ
    \item SegFormer сжимает ключи окнами $R \times R$: $M = N / R^2$, сложность $O(N^2 d / R^2)$, порядок остается квадратичным
    \item Удвоение стороны кадра увеличивает стоимость внимания в 16 раз
    \item SegFormer-B0: 6.8 мс при 256 px и 133.7 мс при 1024 px (7.5 кадров/с)
    \item Линейное внимание~\cite{katharopoulos,flatten} снижает сложность до $O(N d^2)$, но веса внимания становятся размытыми, и модель хуже выделяет мелкие детали и границы объектов
    \item Известные модели с линейным вниманием обучаются с нуля, веса уже обученного SegFormer в них перенести нельзя
  \end{itemize}
\end{frame}

% - Постановка задачи -
\begin{frame}{Постановка задачи}
  \small
  \begin{itemize}
    \item Заменить механизм внимания ESA в каждом блоке энкодера SegFormer-B0 на внимание с линейной сложностью, остальную сеть не менять
    \item Время работы нового внимания растет линейно с числом токенов $N$
    \item Новое внимание использует те же матрицы весов, что и ESA, поэтому веса обученной модели переносятся без изменений, новых параметров не более 1\%
    \item mIoU снижается не более чем на $\varepsilon = 0.005$
    \item Задержка на изображениях высокого разрешения уменьшается минимум в 2 раза
  \end{itemize}
  $$
  \hat{A}^{*} = \arg\min_{\hat{A}} T_{S}\left(f_{\hat{A}}\right) \quad \text{при} \quad \mathrm{mIoU}\left(f_{\hat{A}}\right) \geq \mathrm{mIoU}\left(f_{A}\right) - \varepsilon
  $$
  \begin{itemize}
    \item $A$ - исходное внимание ESA, $\hat{A}$ - новое внимание, $f_A$ и $f_{\hat{A}}$ - модели с ними, $T_S$ - задержка на кадре со стороной $S$ px
    \item Тестовая задача: Oxford-IIIT Pet, 3 класса - фон, животное, граница
    \item Класс границы - тонкое кольцо вдоль контура, по геометрии близок к рельсам
  \end{itemize}
\end{frame}

% - Метод: ядровая линеаризация -
\begin{frame}{Метод: ядровая линеаризация по оконным ключам}
  \small
  \begin{itemize}
    \item Замена softmax на положительную ядровую функцию: $\mathrm{sim}(q, k) = \varphi(q)^{\top} \varphi(k)$
  \end{itemize}
  $$
  \mathrm{out}_i = \frac{\varphi(q_i)^{\top} S}{\varphi(q_i)^{\top} z}, \qquad S = \varphi(K)^{\top} V \in \mathbb{R}^{d_h \times d_h}, \qquad z = \sum_{j=1}^{M} \varphi(k_j)
  $$
  \begin{itemize}
    \item $S$ и $z$ считаются один раз по $M$ оконным токенам и применяются ко всем $N$ запросам
    \item Фокусированная ядровая функция, $p$ обучаемый, $p \in [1, 8]$:
  \end{itemize}
  $$
  \varphi_p(x) = \left\| x^{+} \right\| \left(\frac{x^{+}}{\left\| x^{+} \right\|}\right)^{p}, \qquad x^{+} = \mathrm{ReLU}(x) + \epsilon
  $$
  \begin{itemize}
    \item Локальная оконная ветвь $5 \times 5$ восстанавливает ранг и детали внутри окна:
  \end{itemize}
  $$
  \mathrm{WKLA}(X) = \left(\frac{\varphi_p(Q)\, S}{\varphi_p(Q)\, z} + \mathrm{DWConv}_{5 \times 5}\left(X W_V\right)\right) W_O
  $$
\end{frame}

  \small
  \begin{itemize}
    \item Два окна: глобальное ядровое внимание по токенам окон $R \times R$ и локальная свертка $5 \times 5$, +0.7\% параметров
    \item Веса $W_Q, W_K, W_V, W_O$ и свертка окон копируются из обученного SegFormer-B0
    \item Перенос внимания: 3 эпохи блоки WKLA учатся повторять выход ESA, корреляция с ESA растет с 0.74 до 0.93
  \end{itemize}

% - Эксперимент -
\begin{frame}{Эксперимент}
  \small
  \begin{itemize}
    \item Oxford-IIIT Pet~\cite{pets}: 3312 изображений для обучения, 3669 тестовых, размер $256 \times 256$
    \item Аугментации: отражение, случайное кадрирование 0.7-1.0, яркость и контраст
    \item Единый рецепт для всех вариантов SegFormer: AdamW, шаг $3 \cdot 10^{-4}$ (энкодер $\times 0.1$), батч 16, 20 эпох, зерно 1337
    \item Варианты: ESA, \ours, WKLA только на стадиях 1-2
    \item Абляции: без переноса внимания, без наследования весов, ядро $\mathrm{elu} + 1$
    \item Внешние ориентиры: DeepLabV3+ (ResNet101)~\cite{deeplab}, U-Net~\cite{unet}
    \item Метрики: IoU по классам, mIoU, mDice, $\mathrm{mAP}_{50:95}$
    \item Скорость: Apple Silicon, PyTorch MPS, batch 1, 8 прогревов и 25 замеров
  \end{itemize}
\end{frame}

% - Выводы -
\begin{frame}{Выводы}
  \small
  \begin{itemize}
    \item Предложено оконное ядровое линейное внимание WKLA для уже обученного SegFormer-B0 без повторного предобучения
    \item Качество в пределах допуска: mIoU 0.8050 против 0.8095 (потеря 0.0045), IoU границы 0.5917 против 0.6004, +0.7\% параметров
    \item Стоимость внимания при 1024 px ниже в 16.9 раза, задержка модели в 2.31 раза, 7.5 $\to$ 17.3 кадров/с
    \item Граница реального времени сдвинута с 640 до 768 px
    \item Главный вклад в качество дает наследование весов, затем фокусированное ядро с локальной ветвью; перенос внимания ускоряет сходимость
    \item Ограничения: на разрешении до 512 px модель медленнее ESA, один набор данных и одна платформа
    \item Перенос на SegFormer B3 и B5, дорожные и железнодорожные сцены, слитое ядро линейного внимания, динамическая зона интереса трамвайных путей~\cite{myarticle}
  \end{itemize}
\end{frame}
