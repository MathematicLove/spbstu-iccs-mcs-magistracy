||--|| связь 1-1  ·  ||--o{ связь 1-N / N-1  ·  N-N показана как две связи 1-N через таблицу-справочник (junction table) 
%% taxi_db — поездки такси в Чикаго
%% 1-1: companies ↔ company_stats, trips ↔ trip_payments
%% 1-N: companies → trips, payment_types → trips,
%%      community_areas → trips (дважды: pickup и dropoff),
%%      companies → company_payment_types, payment_types → company_payment_types
%% N-N: companies ↔ payment_types  (реализовано через company_payment_types)
erDiagram
    COMPANIES       ||--|| COMPANY_STATS : "1-1"
    COMPANIES       ||--o{ TRIPS : "1-N"
    PAYMENT_TYPES   ||--o{ TRIPS : "1-N"
    COMMUNITY_AREAS ||--o{ TRIPS : "1-N (pickup)"
    COMMUNITY_AREAS ||--o{ TRIPS : "1-N (dropoff)"
    TRIPS           ||--|| TRIP_PAYMENTS : "1-1"
    COMPANIES       ||--o{ COMPANY_PAYMENT_TYPES : "1-N"
    PAYMENT_TYPES   ||--o{ COMPANY_PAYMENT_TYPES : "1-N"

    COMPANIES {
        serial company_id PK
        varchar company_name UK
    }
    COMPANY_STATS {
        integer company_id PK "FK -> companies.company_id"
        integer trips_count
        numeric avg_total
        numeric avg_miles
    }
    PAYMENT_TYPES {
        smallserial payment_type_id PK
        varchar payment_name UK "Cash, Credit Card ..."
    }
    COMMUNITY_AREAS {
        smallserial area_id PK
        smallint area_code UK "1..77"
    }
    TRIPS {
        bigserial trip_pk PK
        varchar trip_id
        timestamp trip_start
        timestamp trip_end
        integer trip_seconds
        numeric trip_miles
        integer company_id FK
        smallint payment_type_id FK
        smallint pickup_area_id FK
        smallint dropoff_area_id FK
    }
    TRIP_PAYMENTS {
        bigint trip_pk PK "FK -> trips.trip_pk"
        numeric fare
        numeric tips
        numeric tolls
        numeric extras
        numeric trip_total
    }
    COMPANY_PAYMENT_TYPES {
        integer company_id PK "FK -> companies.company_id"
        smallint payment_type_id PK "FK -> payment_types.payment_type_id"
        integer trips_count
    }
